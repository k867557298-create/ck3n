# Opening briefing for the dedicated 1546 start only.
character_event = {
	id = ottoman_suleiman.0001
	title = ottoman_suleiman.0001.t
	desc = ottoman_suleiman.0001.desc
	picture = GFX_evt_court
	trigger = {
		has_title = e_ottoman_empire
		date >= 1546.01.01
		date < 1546.01.02
	}
	is_triggered_only = yes
	option = {
		name = ottoman_suleiman.0001.a
		add_prestige = 250
	}
}

# In 1546 the court has to preserve the navy after Hayreddin Barbarossa's death.
character_event = {
	id = ottoman_suleiman.0100
	title = ottoman_suleiman.0100.t
	desc = ottoman_suleiman.0100.desc
	picture = GFX_evt_council
	trigger = {
		has_title = e_ottoman_empire
		date >= 1546.07.04
		date < 1547.01.01
		NOT = { has_character_flag = ottoman_barbarossa_legacy_resolved }
	}
	is_triggered_only = yes
	option = {
		name = ottoman_suleiman.0100.a
		set_character_flag = ottoman_barbarossa_legacy_resolved
		add_gold = -150
		add_character_modifier = { modifier = ottoman_divan_reform_modifier years = 15 }
	}
	option = {
		name = ottoman_suleiman.0100.b
		set_character_flag = ottoman_barbarossa_legacy_resolved
		add_prestige = -150
	}
}

# The Mustafa crisis is deliberately a player choice rather than a forced outcome.
character_event = {
	id = ottoman_suleiman.0200
	title = ottoman_suleiman.0200.t
	desc = ottoman_suleiman.0200.desc
	picture = GFX_evt_court
	trigger = {
		has_title = e_ottoman_empire
		date >= 1553.01.01
		date < 1554.01.01
		NOT = { has_character_flag = ottoman_mustafa_crisis_resolved }
	}
	is_triggered_only = yes
	option = {
		name = ottoman_suleiman.0200.a
		set_character_flag = ottoman_mustafa_crisis_resolved
		add_prestige = -300
		add_character_modifier = { modifier = ottoman_mustafa_crisis_modifier years = 10 }
	}
	option = {
		name = ottoman_suleiman.0200.b
		set_character_flag = ottoman_mustafa_crisis_resolved
		add_prestige = 200
	}
}
