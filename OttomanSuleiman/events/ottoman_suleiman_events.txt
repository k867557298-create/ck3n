# Opening briefing for the dedicated 1546 start only.

	id = ottoman_suleiman.0001
	title = ottoman_suleiman.0001.t
	desc = ottoman_suleiman.0001.desc
	picture = GFX_evt_court
	trigger = {
		has_title = e_ottoman_empire
		current_date >= 1546.01.01
		current_date < 1546.01.02
	}
	is_triggered_only = yes
	option = {
		name = ottoman_suleiman.0001.a
		add_prestige = 250
	}
}

# In 1546 the court has to preserve the navy after Hayreddin Barbarossa's death.
character_event = {
	id = ottoman_suleiman.0100
	title = ottoman_suleiman.0100.t
	desc = ottoman_suleiman.0100.desc
	picture = GFX_evt_council
	trigger = {
		has_title = e_ottoman_empire
		current_date >= 1546.07.04
		current_date < 1547.01.01
		NOT = { has_character_flag = ottoman_barbarossa_legacy_resolved }
	}
	is_triggered_only = yes
	option = {
		name = ottoman_suleiman.0100.a
		set_character_flag = ottoman_barbarossa_legacy_resolved

	}
	option = {
		name = ottoman_suleiman.0100.b
		set_character_flag = ottoman_barbarossa_legacy_resolved
		add_prestige = -150
	}
}
